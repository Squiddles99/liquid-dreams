# Grommet Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Grommet (Bradley, 13, bodyboarder only) as a third playable body with:
- a curly ginger mop;
- round glasses shown on land;
- buck teeth;
- shader-drawn freckles, pimples and sunburn;
- a wet/dry look driven by an `onLand` switch.

**Architecture:**
- The scripted Blender + MPFB build (`tools/surfer/`) gains:
  - a `curly` hair style;
  - glasses and teeth meshes;
  - a dark mouth interior;
  - landmarks and seeded pimples written to the manifest.
- The game (`src/surfer/`) gains:
  - a third preset whose quiver is only a bodyboard;
  - an outfit `rashieAndBoardies`;
  - materials for the glasses, lenses and teeth;
  - a skin-detail layer in `bodyMaterial`, fed by the manifest's landmarks;
  - a `wet` uniform that the `onLand` switch drives.
- The stand and the dev panel wire it up.

**Tech Stack:**
- three 0.186.1 WebGPU with TSL node materials, TypeScript, vitest, tweakpane;
- Blender 5.2.2 LTS with MPFB 2.0.17, driven headless by `npm run build:surfers`;
- Node 24.

**Spec:** `docs/superpowers/specs/2026-09-30-grommet-design.md` (read it with this plan). The earlier sub-project's spec,
`docs/superpowers/specs/2026-09-30-surfer-on-the-stand-design.md`, still governs everything Grommet shares with the
other two surfers.

## Global Constraints

- Every asset is CC0 or made by the project (Steam). Grommet uses MPFB CC0 data and our own scripts only; no textures
  or images are embedded in any `.glb`.
- The skeleton contract: the same 23 bones, the same names and parents (`src/surfer/rig.ts` `BONES`/`PARENT`).
  `manifestProblems` must return `[]`.
- Height is **1.52 m ±1 cm**. His triangles total **≤ 40 000**, of which the body is **≤ 24 000** with ≤ 4 materials.
- The preset keys `female` and `male` stay as they are (stored settings and links keep working). The new key is
  `grommet`.
- The glb's node transforms are identity and its positions are metres, y up, feet at 0, +z forward: the geometry space
  equals the manifest's space and the game's world when the surfer's group is at identity. Pinned in Task 4.
- Don't change existing test tolerances. New tests set their own, and every threshold is written into the test.
- The tone is a realistic 13-year-old: goofy but loveable, never a cartoon or a gross-out.
- Commit messages end with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`. Don't push or merge until
  Andrew says so.
- Private reference photos stay in `reference/` (git-ignored). Never commit them.

```ts
const BOARD_ORDER: readonly BoardKind[] = ['thruster', 'stepUp', 'bodyboard'];
/** The boards a preset may ride: its quiver's, in the panel's order (Grommet: only his bodyboard). */
export const boardsFor = (p: SurferPreset): BoardKind[] => BOARD_ORDER.filter((k) => p.quiver[k] !== undefined);
export function boardFor(p: SurferPreset, kind: BoardKind): BoardSpec {
  const dims = p.quiver[kind];
  if (!dims) throw new Error(`${p.name} has no ${kind} in the quiver`);
  return makeBoard(kind, dims);
}
```

```python
def _curl(root, n, centre, eye_z, rng):
    """One springy lock: a loose spiral out from the scalp (7–11 cm), shorter over the forehead so it stops above the
    glasses. The radius opens from the root, and nothing dips inside the scalp."""
    fringe = root.y < centre.y - 0.02 and root.z < eye_z + 0.11
    length = rng.uniform(0.035, 0.055) if fringe else rng.uniform(0.07, 0.11)
    radius, pitch, phase = rng.uniform(0.005, 0.011), rng.uniform(0.012, 0.02), rng.uniform(0, 2 * math.pi)
    noise = _unit(rng)
    d = (n * (0.35 if fringe else 0.75) + DOWN * (0.65 if fringe else 0.25) + noise * 0.25).normalized()
    e1 = d.orthogonal().normalized()
    e2 = d.cross(e1)
    r_min = (root - centre).length + 0.002
    pts = []
    for i in range(10):
        s = length * i / 9
        th = phase + 2 * math.pi * s / pitch
        p = root + n * 0.002 + d * s + (e1 * math.cos(th) + e2 * math.sin(th)) * radius * min(1.0, i / 2)
        q = p - centre
        if q.length < r_min:

        if p.y < centre.y - 0.03 and p.z < eye_z + 0.02:  # in front of the face: stay above the glasses
            p.z = eye_z + 0.02
        pts.append(p)
    return pts

    d = ((base - centre).normalized() + _unit(rng) * 0.6).normalized()
    length = rng.uniform(0.015, 0.03)
    return [base, base + d * length * 0.5, base + d * length]
```

```ts
import { readFileSync } from 'node:fs';

/** A float accessor's values from a .glb's binary chunk (tests only). */
export function glbFloats(path: string, gltf: any, accessorIndex: number): Float32Array {
  const b = readFileSync(path);
  const v = new DataView(b.buffer, b.byteOffset, b.byteLength);
  const jsonLen = v.getUint32(12, true), binStart = 20 + jsonLen + 8;
  const acc = gltf.accessors[accessorIndex], view = gltf.bufferViews[acc.bufferView];
  const comps = { SCALAR: 1, VEC2: 2, VEC3: 3, VEC4: 4 }[acc.type as 'SCALAR'];
  const start = binStart + (view.byteOffset ?? 0) + (acc.byteOffset ?? 0);
  return new Float32Array(b.buffer.slice(b.byteOffset + start, b.byteOffset + start + acc.count * comps * 4));
}
```

```python
"""Grommet's big round glasses (grommet spec §4): thick round rims, a keyhole bridge, arms back over the ears, lenses;
skinned to the head. Blender axes: +x the character's left, -y the front, +z up."""
import math

import bmesh
import bpy
from mathutils import Vector
from mathutils.bvhtree import BVHTree

LENS_R = 0.025   # 5 cm across
RIM_R = 0.0028   # the rim's thickness
SIDES = 8


def _tube(bm, pts, r, closed):
    """Sweep a circle of radius r along pts, the frame carried along each point so it never flips."""
    n, rings, u = len(pts), [], None
    for i, p in enumerate(pts):
        a, b = (pts[(i + 1) % n], pts[i - 1]) if closed else (pts[min(i + 1, n - 1)], pts[max(i - 1, 0)])
        t = (a - b).normalized()
        u = t.orthogonal().normalized() if u is None else (u - t * u.dot(t)).normalized()
        v = t.cross(u)
        rings.append([bm.verts.new(p + (u * math.cos(2 * math.pi * k / SIDES) + v * math.sin(2 * math.pi * k / SIDES)) * r) for k in range(SIDES)])
    for i in range(n if closed else n - 1):
        r0, r1 = rings[i], rings[(i + 1) % n]
        for k in range(SIDES):
            bm.faces.new((r0[k], r0[(k + 1) % SIDES], r1[(k + 1) % SIDES], r1[k]))
    if not closed:
        bm.faces.new(list(reversed(rings[0])))
        bm.faces.new(rings[-1])


def build(rig, body, L, name):
    eyes, ears = L["eyes"], L["ears"]
    centres = {s: eyes[s] + Vector((0, -0.0235, 0.002)) for s in ("l", "r")}  # ~12 mm in front of the cornea
    bm = bmesh.new()
    lens_faces = []
    for s, sign in (("l", 1), ("r", -1)):
        c = centres[s]
        rim = [c + Vector((math.cos(a) * LENS_R, 0, math.sin(a) * LENS_R)) for a in (2 * math.pi * k / 32 for k in range(32))]
        _tube(bm, rim, RIM_R, True)
        # The lens: a fan, bulged 2 mm forward in the middle.
        mid = bm.verts.new(c + Vector((0, -0.002, 0)))
        ring = [bm.verts.new(c + Vector((math.cos(a) * LENS_R * 0.97, 0, math.sin(a) * LENS_R * 0.97))) for a in (2 * math.pi * k / 32 for k in range(32))]
        for k in range(32):
            lens_faces.append(bm.faces.new((mid, ring[(k + 1) % 32], ring[k])))
        # The arm: from the rim's outer edge, out past the temple, back to just above the ear, then down behind it.
        start = c + Vector((sign * LENS_R, 0, 0.008))
        ear = ears[s]
        temple = start.lerp(ear, 0.45) + Vector((sign * 0.006, 0, 0.004))
        top = ear + Vector((-sign * 0.004, 0.005, 0.012))
        behind = ear + Vector((-sign * 0.006, 0.022, -0.015))
        _tube(bm, [start, temple, start.lerp(top, 0.8), top, top.lerp(behind, 0.5), behind], RIM_R * 0.8, False)
    # The keyhole bridge: an arc between the rims' inner edges, lifted 4 mm in the middle.
    a, b = centres["r"] + Vector((LENS_R, 0, 0.012)), centres["l"] + Vector((-LENS_R, 0, 0.012))
    _tube(bm, [a.lerp(b, s) + Vector((0, -0.002, 0.004 * math.sin(math.pi * s))) for s in (i / 8 for i in range(9))], RIM_R * 0.9, False)
    me = bpy.data.meshes.new(f"{name}_glasses")
    bm.to_mesh(me)
    bm.free()
    frame_mat = bpy.data.materials.get("glasses") or bpy.data.materials.new("glasses")
    lens_mat = bpy.data.materials.get("lens") or bpy.data.materials.new("lens")
    for m in (frame_mat, lens_mat):
        m.use_nodes = True
        me.materials.append(m)
    for poly, i in zip(me.polygons, range(len(me.polygons))):
        poly.material_index = 0
    lens_ids = _lens_polys(me)
    for i in lens_ids:
        me.polygons[i].material_index = 1
    obj = bpy.data.objects.new(f"{name}_glasses", me)
    bpy.context.scene.collection.objects.link(obj)
    _check_outside(obj, body)
    obj.vertex_groups.new(name="head").add(range(len(me.vertices)), 1.0, "REPLACE")
    obj.modifiers.new("Armature", "ARMATURE").object = rig
    obj.parent = rig
    return obj


def _lens_polys(me):
    """The lens fans are the triangles whose three corners sit within 1 mm of one plane y = const (the rims are tubes)."""
    out = []
    for p in me.polygons:
        ys = [me.vertices[v].co.y for v in p.vertices]
        if len(p.vertices) == 3 and max(ys) - min(ys) < 0.0025 + 1e-6 and p.area > 1e-6:
            out.append(p.index)
    return out


def _check_outside(obj, body):
    """Fail the build if any frame vertex sits inside his head (spec §7)."""
    tree = BVHTree.FromObject(body, bpy.context.evaluated_depsgraph_get())
    for v in obj.data.vertices:
        hit, normal, _, _ = tree.find_nearest(v.co)
        if hit is not None and (v.co - hit).dot(normal) < -0.001:
            raise SystemExit(f"the glasses go inside the head at {tuple(round(c, 3) for c in v.co)}")
```

```python
4 mm, splayed a touch; skinned to the head. Blender axes: +x the character's left, -y the front, +z up."""
import math

import bmesh
import bpy
from mathutils import Matrix, Vector

# (half-spacing from the midline, width, height, depth, forward tilt °, splay °): centrals, laterals, canines.
TEETH = [(0.0048, 0.0095, 0.012, 0.0035, 12, 4), (0.0135, 0.0075, 0.0095, 0.0032, 6, 8), (0.0205, 0.0072, 0.0098, 0.0038, 3, 14)]
ARCH = 18.0  # how fast the arch curves back (m⁻¹): y back = ARCH · x²


def build(rig, L, name):
    ut = L.get("upper_teeth")
    if ut is None:
        raise SystemExit("no upper-teeth helper found: can't place the teeth")
    front, centre = ut["front"], ut["centre"]
    gum_z = centre.z + 0.004
    bm = bmesh.new()
    for x, w, h, d, tilt, splay in TEETH:
        for sign in (1, -1):
            geom = bmesh.ops.create_cube(bm, size=1.0)
            verts = geom["verts"]
            bmesh.ops.scale(bm, vec=Vector((w, d, h)), verts=verts)
            bmesh.ops.translate(bm, vec=Vector((0, 0, -h / 2)), verts=verts)  # hang from the gum line
            rot = Matrix.Rotation(math.radians(-tilt), 3, "X") @ Matrix.Rotation(math.radians(splay * sign), 3, "Z")
            bmesh.ops.rotate(bm, cent=Vector(), matrix=rot, verts=verts)
            pos = Vector((centre.x + sign * x, front.y - 0.004 + ARCH * x * x, gum_z))
            bmesh.ops.translate(bm, vec=pos, verts=verts)
    bmesh.ops.bevel(bm, geom=list(bm.edges), offset=0.0008, segments=1, affect="EDGES")
    me = bpy.data.meshes.new(f"{name}_teeth")
    bm.to_mesh(me)
    bm.free()
    obj = bpy.data.objects.new(f"{name}_teeth", me)
    bpy.context.scene.collection.objects.link(obj)
    obj.vertex_groups.new(name="head").add(range(len(me.vertices)), 1.0, "REPLACE")
    obj.modifiers.new("Armature", "ARMATURE").object = rig
    obj.parent = rig
    return obj
```

```ts
/** The luminance spread across a close-up of the cheek: freckles make it speckled (grommet spec §7). */
async function cheekSpread(renderer: THREE.WebGPURenderer, s: Surfer): Promise<number> {
  const scene = new THREE.Scene();
  scene.add(s.group);
  s.group.updateMatrixWorld(true);
  const eye = s.rest.joint.head.clone().add(new THREE.Vector3(0.035, 0.05, 0.06));
  const cam = new THREE.PerspectiveCamera(12, 1, 0.01, 5);
  cam.position.copy(eye.clone().add(new THREE.Vector3(0.05, 0, 0.35)));
  cam.lookAt(eye);
  const target = new THREE.RenderTarget(96, 96);
  renderer.setRenderTarget(target);
  renderer.render(scene, cam);
  renderer.setRenderTarget(null);
  const px = await renderer.readRenderTargetPixelsAsync(target, 24, 24, 48, 48);
  target.dispose();
  const lum: number[] = [];
  for (let i = 0; i < px.length; i += 4) if (px[i + 3] > 0) lum.push(0.2126 * px[i] + 0.7152 * px[i + 1] + 0.0722 * px[i + 2]);
  const mean = lum.reduce((a, b) => a + b, 0) / Math.max(1, lum.length);
  return Math.sqrt(lum.reduce((a, b) => a + (b - mean) ** 2, 0) / Math.max(1, lum.length));
}
```

```ts
import type { SurferManifest } from './rig';

type V3 = [number, number, number];
/** Where freckles and sunburn gather, in the body's geometry space (metres, glTF axes, rest pose; grommet spec §5). */
export interface SkinZones {
  cheeks: [V3, V3];
  bridge: V3;
  forehead: V3;
  earTops: [V3, V3];
  shoulders: [V3, V3];
  /** Elbow → wrist, each side. */
  forearms: [[V3, V3], [V3, V3]];
  pimples: [number, number, number, number][];
}

const add = (a: V3, b: V3): V3 => [a[0] + b[0], a[1] + b[1], a[2] + b[2]];

/** The zones, or null when the build wrote no landmarks (the others' bodies: no skin detail, nothing breaks). */
export function skinZones(m: SurferManifest): SkinZones | null {
  const L = m.landmarks;
  if (!L) return null;
  const bone = (n: string): V3 => m.bones.find((b) => b.name === n)!.head as V3;
  const side = (i: 0 | 1): number => (i === 0 ? 1 : -1);
  const cheek = (i: 0 | 1): V3 => add(L.eyes[i], [0.012 * side(i), -0.025, 0.004]);
  return {
    cheeks: [cheek(0), cheek(1)],
    bridge: add([(L.eyes[0][0] + L.eyes[1][0]) / 2, (L.eyes[0][1] + L.nose[1]) / 2, (L.eyes[0][2] + L.nose[2]) / 2], [0, 0, 0.004]),
    forehead: [0, L.eyes[0][1] + 0.045, L.eyes[0][2] - 0.004],
    earTops: [add(L.ears[0], [0, 0.015, 0]), add(L.ears[1], [0, 0.015, 0])],
    shoulders: [bone('upperarm_l'), bone('upperarm_r')],
    forearms: [[bone('forearm_l'), bone('hand_l')], [bone('forearm_r'), bone('hand_r')]],
    pimples: m.skin?.pimples ?? [],
  };
}
```

```ts
function toSegment(P: N, a: [number, number, number], b: [number, number, number]): N {
  const A = vec3(...a), AB = vec3(...b).sub(A);

  }
  return out;
```
